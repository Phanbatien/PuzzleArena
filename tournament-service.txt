using Microsoft.AspNetCore.SignalR;
using Microsoft.EntityFrameworkCore;
using PuzzleArena.Api.Data;
using PuzzleArena.Api.Domain;
using PuzzleArena.Api.Games;
using PuzzleArena.Api.Hubs;

namespace PuzzleArena.Api.Services;

public record Standing(int Rank, int UserId, string Username, string Status, int Progress, int Mistakes, int? DurationMs, bool IsFinalist);

/// <summary>
/// Two rounds, as agreed with the lecturer: an open qualifier (everyone, same puzzle, same start)
/// then a final for the fastest N on a different puzzle.
/// </summary>
public class TournamentService(AppDb db, AttemptService attempts, IHubContext<ArenaHub> hub)
{
    public Task<Tournament?> LoadAsync(int id) => db.Tournaments
        .Include(t => t.Entries).ThenInclude(e => e.User)
        .Include(t => t.Puzzles).ThenInclude(p => p.Puzzle)
        .FirstOrDefaultAsync(t => t.Id == id);

    /// <summary>The leaderboard is computed, never stored: solved first (time, then mistakes), then by progress.</summary>
    public static List<Standing> Rank(IEnumerable<TournamentEntry> entries, IEnumerable<Attempt> stageAttempts)
    {
        var byUser = stageAttempts.ToDictionary(a => a.UserId);
        return entries
            .Select(e => (Entry: e, Attempt: byUser.GetValueOrDefault(e.UserId)))
            .OrderBy(x => x.Attempt?.Status == AttemptStatus.Solved ? 0 : 1)
            .ThenBy(x => x.Attempt?.Status == AttemptStatus.Solved ? x.Attempt.DurationMs : 0)
            .ThenBy(x => x.Attempt?.Status == AttemptStatus.Solved ? x.Attempt.Mistakes : 0)
            .ThenByDescending(x => x.Attempt?.Progress ?? -1)
            .ThenBy(x => x.Entry.RegisteredAt)
            .Select((x, i) => new Standing(i + 1, x.Entry.UserId, x.Entry.User.Username,
                x.Attempt?.Status.ToString() ?? "NotStarted", x.Attempt?.Progress ?? 0, x.Attempt?.Mistakes ?? 0,
                x.Attempt?.DurationMs, x.Entry.IsFinalist))
            .ToList();
    }

    public static TournamentPuzzle? StagePuzzle(Tournament t, Stage stage) => t.Puzzles.FirstOrDefault(p => p.Stage == stage);

    /// <summary>Moves the tournament to its next status. Returns an error message, or null on success.</summary>
    public async Task<string?> AdvanceAsync(Tournament t)
    {
        var now = DateTime.UtcNow;
        switch (t.Status)
        {
            case TournamentStatus.Open:
                if (t.Entries.Count == 0) return "Ch╞░a c├│ ng╞░ß╗¥i ch╞íi n├áo ─æ─âng k├╜.";
                AddPuzzle(t, Stage.Qualifier);
                t.QualifierStartedAt = now;
                t.Status = TournamentStatus.Qualifier;
                break;

            case TournamentStatus.Qualifier:
                var qualified = (await CloseStageAsync(t, Stage.Qualifier, t.Entries))
                    .Where(s => s.Status != "NotStarted").Take(t.FinalistCount).Select(s => s.UserId).ToHashSet();
                foreach (var e in t.Entries) e.IsFinalist = qualified.Contains(e.UserId);
                t.Status = TournamentStatus.QualifierDone;
                break;

            case TournamentStatus.QualifierDone:
                if (!t.Entries.Any(e => e.IsFinalist)) return "Kh├┤ng c├│ ai v╞░ß╗út qua v├▓ng loß║íi.";
                AddPuzzle(t, Stage.Final);
                t.FinalStartedAt = now;
                t.Status = TournamentStatus.Final;
                break;

            case TournamentStatus.Final:
                var finalists = t.Entries.Where(e => e.IsFinalist).ToList();
                foreach (var s in await CloseStageAsync(t, Stage.Final, finalists))
                    finalists.First(e => e.UserId == s.UserId).FinalRank = s.Rank;
                t.Status = TournamentStatus.Closed;
                break;

            default:
                return "Giß║úi ─æß║Ñu ─æ├ú kß║┐t th├║c.";
        }
        await db.SaveChangesAsync();
        await hub.Clients.Group(ArenaHub.Public(t.Id)).SendAsync("stage", t.Status);
        return null;
    }

    /// <summary>The puzzle is generated only when the round starts, so nobody (admins included) can see it early.</summary>
    static void AddPuzzle(Tournament t, Stage stage) => t.Puzzles.Add(new TournamentPuzzle
    {
        Stage = stage,
        Puzzle = GameEngine.Generate(t.GameType, t.Difficulty, Random.Shared.Next()),
    });

    /// <summary>Times out whoever is still playing, then ranks the stage.</summary>
    async Task<List<Standing>> CloseStageAsync(Tournament t, Stage stage, IEnumerable<TournamentEntry> entries)
    {
        var puzzleId = StagePuzzle(t, stage)!.PuzzleId;
        var stageAttempts = await db.Attempts.Include(a => a.Puzzle).Where(a => a.PuzzleId == puzzleId).ToListAsync();
        foreach (var a in stageAttempts.Where(a => a.Status == AttemptStatus.InProgress))
            await attempts.FinishAsync(a, AttemptStatus.Failed);
        return Rank(entries, stageAttempts);
    }
}
